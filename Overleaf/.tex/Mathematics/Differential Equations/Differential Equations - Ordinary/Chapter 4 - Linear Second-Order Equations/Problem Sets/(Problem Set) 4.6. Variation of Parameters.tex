\section{Concept Checklist \& Question Mapping}
This section outlines the core concepts, problem archetypes, and common tricks found in the Variation of Parameters method, mapped to the corresponding practice problems.

\begin{itemize}
    \item \textbf{Concept 1: Basic Trigonometric Nonhomogeneous Terms} \\
    \textit{Description:} Involves $f(t)$ like $\tan(t)$, $\sec(t)$, $\csc(t)$, or $\cot(t)$. Requires trigonometric identities to simplify $v_1'$ and $v_2'$ before integrating. \\
    \textit{Questions:} 1, 2, 3, 4, 5, 6
    
    \item \textbf{Concept 2: Rational/Exponential Combinations} \\
    \textit{Description:} Involves $f(t)$ like $e^t/t$ or $e^{2t}/(1+t^2)$. The Wronskian typically cancels the exponential part, leaving a rational function to integrate (often resulting in natural logarithms or inverse tangents). \\
    \textit{Questions:} 7, 8, 9, 10, 11
    
    \item \textbf{Concept 3: The ``a'' Check (Leading Coefficient $\neq 1$)} \\
    \textit{Description:} The standard form requires $y''$ to have a coefficient of 1. If the equation is $ay'' + by' + cy = f(t)$, you must divide $f(t)$ by $a$ in the parameter formulas. \\
    \textit{Questions:} 12, 13, 14, 15
    
    \item \textbf{Concept 4: Superposition Principle} \\
    \textit{Description:} Splitting a complex $f(t)$ into simpler parts. Uses Undetermined Coefficients for polynomials/exponentials/sines/cosines, and Variation of Parameters for the rest, saving time. \\
    \textit{Questions:} 16, 17, 18, 19
    
    \item \textbf{Concept 5: Definite Integrals, Abstract Functions, and IVPs} \\
    \textit{Description:} When $f(t)$ has no elementary antiderivative, definite integrals from $t_0$ to $t$ are used. Relies on the Fundamental Theorem of Calculus for Initial Value Problems. \\
    \textit{Questions:} 20, 21, 22, 23
    
    \item \textbf{Concept 6: Variable Coefficient Equations (Cauchy-Euler)} \\
    \textit{Description:} Equations of the form $t^2y'' + aty' + by = f(t)$. Requires dividing by $t^2$ to achieve standard form before applying Variation of Parameters. \\
    \textit{Questions:} 24, 25, 26, 27
    
    \item \textbf{Concept 7: Advanced Diagnostic Testing (Find the Flaw)} \\
    \textit{Description:} Identifying common pitfalls such as forgetting the negative sign in $v_1'$, forgetting to divide by $a$, or using an incorrect Wronskian formula. \\
    \textit{Questions:} 28, 29, 30
\end{itemize}

\section{Problem Sets}

\subsection*{Concept 1: Basic Trigonometric Nonhomogeneous Terms}
Find a general solution to the following differential equations using Variation of Parameters.
\begin{enumerate}
    \item $y'' + y = \tan(t)$
    \item $y'' + 9y = \sec(3t)$
    \item $y'' + 4y = \csc(2t)$
    \item $y'' + y = \sec(t)\tan(t)$
    \item $y'' + 16y = \sec^2(4t)$
    \item $y'' + y = \cot(t)$
\end{enumerate}

\subsection*{Concept 2: Rational/Exponential Combinations}
Find a general solution to the following differential equations.
\begin{enumerate}
    \setcounter{enumi}{6}
    \item $y'' - 4y' + 4y = \frac{e^{2t}}{t}$
    \item $y'' - 2y' + y = \frac{e^t}{t^2}$
    \item $y'' + 2y' + y = e^{-t}\ln(t)$
    \item $y'' - 6y' + 9y = \frac{e^{3t}}{t^3}$
    \item $y'' - 4y' + 4y = \frac{e^{2t}}{1+t^2}$
\end{enumerate}

\subsection*{Concept 3: The ``a'' Check (Leading Coefficient $\neq 1$)}
Find a general solution to the following differential equations. Pay attention to the leading coefficient.
\begin{enumerate}
    \setcounter{enumi}{11}
    \item $2y'' + 2y = \tan(t)$
    \item $3y'' - 6y' + 3y = \frac{e^t}{t}$
    \item $4y'' + 16y = \sec(2t)$
    \item $2y'' - 8y' + 8y = \frac{e^{2t}}{t^2}$
\end{enumerate}

\subsection*{Concept 4: Superposition Principle}
Find a general solution to the following differential equations by splitting the nonhomogeneous term and using the most efficient method for each part.
\begin{enumerate}
    \setcounter{enumi}{15}
    \item $y'' + y = \sec(t) + e^{2t}$
    \item $y'' - 2y' + y = \frac{e^t}{t} + 3t$
    \item $y'' + 4y = \tan(2t) + 5e^{-t}$
    \item $y'' + 9y = \csc(3t) + \sin(t)$
\end{enumerate}

\subsection*{Concept 5: Definite Integrals, Abstract Functions, and IVPs}
\begin{enumerate}
    \setcounter{enumi}{19}
    \item Express the solution to the IVP $y'' - y = \frac{1}{t^2}$, $y(1) = 0$, $y'(1) = 0$ using definite integrals.
    \item Express the solution to the IVP $y'' + y = e^{-t^2}$, $y(0) = 0$, $y'(0) = 0$ using definite integrals.
    \item Show that $y_p(t) = \int_0^t f(s) \frac{\sin(2(t-s))}{2} \, ds$ is a particular solution to $y'' + 4y = f(t)$.
    \item Show that $y_p(t) = \int_0^t f(s) (t-s)e^{t-s} \, ds$ is a particular solution to $y'' - 2y' + y = f(t)$.
\end{enumerate}

\subsection*{Concept 6: Variable Coefficient Equations (Cauchy-Euler)}
Find a particular solution to the following variable coefficient equations, given the homogeneous solutions $y_1$ and $y_2$.
\begin{enumerate}
    \setcounter{enumi}{23}
    \item $t^2 y'' - 2y = t^3$, given $y_1 = t^2$ and $y_2 = t^{-1}$.
    \item $t^2 y'' + t y' - y = t^2$, given $y_1 = t$ and $y_2 = t^{-1}$.
    \item $t^2 y'' - 3t y' + 3y = t^4$, given $y_1 = t$ and $y_2 = t^3$.
    \item $t^2 y'' + 2t y' - 2y = t^{-1}$, given $y_1 = t$ and $y_2 = t^{-2}$.
\end{enumerate}

\subsection*{Concept 7: Advanced Diagnostic Testing (Find the Flaw)}
Identify the error in the provided flawed solutions and write the correct particular solution $y_p$.
\begin{enumerate}
    \setcounter{enumi}{27}
    \item \textbf{Problem:} Find a particular solution to $y'' + y = \sec(t)$. \\
    \textbf{Flawed Solution:} 1. $y_1 = \cos(t), y_2 = \sin(t), W = 1$. 2. $v_1' = \sin(t)\sec(t) = \tan(t) \implies v_1 = -\ln|\cos(t)|$. 3. $v_2' = \cos(t)\sec(t) = 1 \implies v_2 = t$. 4. $y_p = -\cos(t)\ln|\cos(t)| + t\sin(t)$.
    
    \item \textbf{Problem:} Find a particular solution to $3y'' + 3y = \tan(t)$. \\
    \textbf{Flawed Solution:} 1. $y_1 = \cos(t), y_2 = \sin(t), W = 1$. 2. $v_1' = -\sin(t)\tan(t)$. 3. $v_2' = \cos(t)\tan(t)$. 4. $y_p = -\cos(t)\ln|\sec(t)+\tan(t)|$.
    
    \item \textbf{Problem:} Find a particular solution to $y'' - 2y' + y = \frac{e^t}{t}$. \\
    \textbf{Flawed Solution:} 1. $y_1 = e^t, y_2 = te^t$. 2. $W = y_1 y_2' + y_1' y_2 = e^t(e^t + te^t) + e^t(te^t) = e^{2t} + 2te^{2t}$. 3. $v_1' = \frac{-te^t(e^t/t)}{e^{2t} + 2te^{2t}}$.
\end{enumerate}

\section{Detailed Solutions}

\subsection*{Concept 1 Solutions}
\textbf{1.} $y'' + y = \tan(t)$ \\
$y_c = c_1\cos(t) + c_2\sin(t)$. $y_1 = \cos(t)$, $y_2 = \sin(t)$, $W = 1$. \\
$v_1' = -\sin(t)\tan(t) = -\frac{\sin^2(t)}{\cos(t)} = \cos(t) - \sec(t) \implies v_1 = \sin(t) - \ln|\sec(t) + \tan(t)|$. \\
$v_2' = \cos(t)\tan(t) = \sin(t) \implies v_2 = -\cos(t)$. \\
$y_p = \cos(t)(\sin(t) - \ln|\sec(t) + \tan(t)|) - \sin(t)\cos(t) = -\cos(t)\ln|\sec(t) + \tan(t)|$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(t) + c_2\sin(t) - \cos(t)\ln|\sec(t) + \tan(t)|$

\vspace{0.5cm}
\textbf{2.} $y'' + 9y = \sec(3t)$ \\
$y_c = c_1\cos(3t) + c_2\sin(3t)$. $W = 3$. \\
$v_1' = -\frac{\sin(3t)\sec(3t)}{3} = -\frac{1}{3}\tan(3t) \implies v_1 = \frac{1}{9}\ln|\cos(3t)|$. \\
$v_2' = \frac{\cos(3t)\sec(3t)}{3} = \frac{1}{3} \implies v_2 = \frac{1}{3}t$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(3t) + c_2\sin(3t) + \frac{1}{9}\cos(3t)\ln|\cos(3t)| + \frac{1}{3}t\sin(3t)$

\vspace{0.5cm}
\textbf{3.} $y'' + 4y = \csc(2t)$ \\
$y_c = c_1\cos(2t) + c_2\sin(2t)$. $W = 2$. \\
$v_1' = -\frac{\sin(2t)\csc(2t)}{2} = -\frac{1}{2} \implies v_1 = -\frac{1}{2}t$. \\
$v_2' = \frac{\cos(2t)\csc(2t)}{2} = \frac{1}{2}\cot(2t) \implies v_2 = \frac{1}{4}\ln|\sin(2t)|$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(2t) + c_2\sin(2t) - \frac{1}{2}t\cos(2t) + \frac{1}{4}\sin(2t)\ln|\sin(2t)|$

\vspace{0.5cm}
\textbf{4.} $y'' + y = \sec(t)\tan(t)$ \\
$y_c = c_1\cos(t) + c_2\sin(t)$. $W = 1$. \\
$v_1' = -\sin(t)\sec(t)\tan(t) = -\tan^2(t) = 1 - \sec^2(t) \implies v_1 = t - \tan(t)$. \\
$v_2' = \cos(t)\sec(t)\tan(t) = \tan(t) \implies v_2 = \ln|\sec(t)|$. \\
$y_p = (t - \tan(t))\cos(t) + \sin(t)\ln|\sec(t)| = t\cos(t) - \sin(t) + \sin(t)\ln|\sec(t)|$. (The $-\sin(t)$ absorbs into $y_c$). \\
\textbf{Final Answer:} $y(t) = c_1\cos(t) + c_2\sin(t) + t\cos(t) + \sin(t)\ln|\sec(t)|$

\vspace{0.5cm}
\textbf{5.} $y'' + 16y = \sec^2(4t)$ \\
$y_c = c_1\cos(4t) + c_2\sin(4t)$. $W = 4$. \\
$v_1' = -\frac{\sin(4t)\sec^2(4t)}{4} = -\frac{1}{4}\tan(4t)\sec(4t) \implies v_1 = -\frac{1}{16}\sec(4t)$. \\
$v_2' = \frac{\cos(4t)\sec^2(4t)}{4} = \frac{1}{4}\sec(4t) \implies v_2 = \frac{1}{16}\ln|\sec(4t) + \tan(4t)|$. \\
$y_p = -\frac{1}{16}\sec(4t)\cos(4t) + \frac{1}{16}\sin(4t)\ln|\sec(4t) + \tan(4t)| = -\frac{1}{16} + \frac{1}{16}\sin(4t)\ln|\sec(4t) + \tan(4t)|$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(4t) + c_2\sin(4t) - \frac{1}{16} + \frac{1}{16}\sin(4t)\ln|\sec(4t) + \tan(4t)|$

\vspace{0.5cm}
\textbf{6.} $y'' + y = \cot(t)$ \\
$y_c = c_1\cos(t) + c_2\sin(t)$. $W = 1$. \\
$v_1' = -\sin(t)\cot(t) = -\cos(t) \implies v_1 = -\sin(t)$. \\
$v_2' = \cos(t)\cot(t) = \frac{\cos^2(t)}{\sin(t)} = \csc(t) - \sin(t) \implies v_2 = -\ln|\csc(t) + \cot(t)| + \cos(t)$. \\
$y_p = -\sin(t)\cos(t) - \sin(t)\ln|\csc(t) + \cot(t)| + \sin(t)\cos(t) = -\sin(t)\ln|\csc(t) + \cot(t)|$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(t) + c_2\sin(t) - \sin(t)\ln|\csc(t) + \cot(t)|$

\subsection*{Concept 2 Solutions}
\textbf{7.} $y'' - 4y' + 4y = \frac{e^{2t}}{t}$ \\
$y_c = c_1e^{2t} + c_2te^{2t}$. $W = e^{4t}$. \\
$v_1' = -\frac{te^{2t}(e^{2t}/t)}{e^{4t}} = -1 \implies v_1 = -t$. \\
$v_2' = \frac{e^{2t}(e^{2t}/t)}{e^{4t}} = \frac{1}{t} \implies v_2 = \ln|t|$. \\
$y_p = -te^{2t} + te^{2t}\ln|t|$. (The $-te^{2t}$ absorbs into $y_c$). \\
\textbf{Final Answer:} $y(t) = c_1e^{2t} + c_2te^{2t} + te^{2t}\ln|t|$

\vspace{0.5cm}
\textbf{8.} $y'' - 2y' + y = \frac{e^t}{t^2}$ \\
$y_c = c_1e^t + c_2te^t$. $W = e^{2t}$. \\
$v_1' = -\frac{te^t(e^t/t^2)}{e^{2t}} = -\frac{1}{t} \implies v_1 = -\ln|t|$. \\
$v_2' = \frac{e^t(e^t/t^2)}{e^{2t}} = \frac{1}{t^2} \implies v_2 = -\frac{1}{t}$. \\
$y_p = -e^t\ln|t| - e^t$. (The $-e^t$ absorbs into $y_c$). \\
\textbf{Final Answer:} $y(t) = c_1e^t + c_2te^t - e^t\ln|t|$

\vspace{0.5cm}
\textbf{9.} $y'' + 2y' + y = e^{-t}\ln(t)$ \\
$y_c = c_1e^{-t} + c_2te^{-t}$. $W = e^{-2t}$. \\
$v_1' = -\frac{te^{-t}(e^{-t}\ln t)}{e^{-2t}} = -t\ln(t) \implies v_1 = -\frac{t^2}{2}\ln(t) + \frac{t^2}{4}$ (via integration by parts). \\
$v_2' = \frac{e^{-t}(e^{-t}\ln t)}{e^{-2t}} = \ln(t) \implies v_2 = t\ln(t) - t$. \\
$y_p = e^{-t}(-\frac{t^2}{2}\ln(t) + \frac{t^2}{4}) + te^{-t}(t\ln(t) - t) = e^{-t}(\frac{t^2}{2}\ln(t) - \frac{3t^2}{4})$. \\
\textbf{Final Answer:} $y(t) = c_1e^{-t} + c_2te^{-t} + e^{-t}(\frac{t^2}{2}\ln(t) - \frac{3t^2}{4})$

\vspace{0.5cm}
\textbf{10.} $y'' - 6y' + 9y = \frac{e^{3t}}{t^3}$ \\
$y_c = c_1e^{3t} + c_2te^{3t}$. $W = e^{6t}$. \\
$v_1' = -\frac{te^{3t}(e^{3t}/t^3)}{e^{6t}} = -\frac{1}{t^2} \implies v_1 = \frac{1}{t}$. \\
$v_2' = \frac{e^{3t}(e^{3t}/t^3)}{e^{6t}} = \frac{1}{t^3} \implies v_2 = -\frac{1}{2t^2}$. \\
$y_p = \frac{1}{t}e^{3t} - \frac{1}{2t^2}te^{3t} = \frac{e^{3t}}{2t}$. \\
\textbf{Final Answer:} $y(t) = c_1e^{3t} + c_2te^{3t} + \frac{e^{3t}}{2t}$

\vspace{0.5cm}
\textbf{11.} $y'' - 4y' + 4y = \frac{e^{2t}}{1+t^2}$ \\
$y_c = c_1e^{2t} + c_2te^{2t}$. $W = e^{4t}$. \\
$v_1' = -\frac{te^{2t}(e^{2t}/(1+t^2))}{e^{4t}} = -\frac{t}{1+t^2} \implies v_1 = -\frac{1}{2}\ln(1+t^2)$. \\
$v_2' = \frac{e^{2t}(e^{2t}/(1+t^2))}{e^{4t}} = \frac{1}{1+t^2} \implies v_2 = \arctan(t)$. \\
\textbf{Final Answer:} $y(t) = c_1e^{2t} + c_2te^{2t} - \frac{1}{2}e^{2t}\ln(1+t^2) + te^{2t}\arctan(t)$

\subsection*{Concept 3 Solutions}
\textbf{12.} $2y'' + 2y = \tan(t)$ \\
Divide by $a=2$: $y'' + y = \frac{1}{2}\tan(t)$. $W = 1$. \\
$v_1' = -\frac{1}{2}\sin(t)\tan(t) \implies v_1 = \frac{1}{2}(\sin(t) - \ln|\sec(t) + \tan(t)|)$. \\
$v_2' = \frac{1}{2}\cos(t)\tan(t) = \frac{1}{2}\sin(t) \implies v_2 = -\frac{1}{2}\cos(t)$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(t) + c_2\sin(t) - \frac{1}{2}\cos(t)\ln|\sec(t) + \tan(t)|$

\vspace{0.5cm}
\textbf{13.} $3y'' - 6y' + 3y = \frac{e^t}{t}$ \\
Divide by $a=3$: $y'' - 2y' + y = \frac{e^t}{3t}$. $W = e^{2t}$. \\
$v_1' = -\frac{te^t(e^t/(3t))}{e^{2t}} = -\frac{1}{3} \implies v_1 = -\frac{t}{3}$. \\
$v_2' = \frac{e^t(e^t/(3t))}{e^{2t}} = \frac{1}{3t} \implies v_2 = \frac{1}{3}\ln|t|$. \\
\textbf{Final Answer:} $y(t) = c_1e^t + c_2te^t + \frac{1}{3}te^t\ln|t|$

\vspace{0.5cm}
\textbf{14.} $4y'' + 16y = \sec(2t)$ \\
Divide by $a=4$: $y'' + 4y = \frac{1}{4}\sec(2t)$. $W = 2$. \\
$v_1' = -\frac{\sin(2t)(\frac{1}{4}\sec(2t))}{2} = -\frac{1}{8}\tan(2t) \implies v_1 = \frac{1}{16}\ln|\cos(2t)|$. \\
$v_2' = \frac{\cos(2t)(\frac{1}{4}\sec(2t))}{2} = \frac{1}{8} \implies v_2 = \frac{t}{8}$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(2t) + c_2\sin(2t) + \frac{1}{16}\cos(2t)\ln|\cos(2t)| + \frac{t}{8}\sin(2t)$

\vspace{0.5cm}
\textbf{15.} $2y'' - 8y' + 8y = \frac{e^{2t}}{t^2}$ \\
Divide by $a=2$: $y'' - 4y' + 4y = \frac{e^{2t}}{2t^2}$. $W = e^{4t}$. \\
$v_1' = -\frac{te^{2t}(e^{2t}/(2t^2))}{e^{4t}} = -\frac{1}{2t} \implies v_1 = -\frac{1}{2}\ln|t|$. \\
$v_2' = \frac{e^{2t}(e^{2t}/(2t^2))}{e^{4t}} = \frac{1}{2t^2} \implies v_2 = -\frac{1}{2t}$. \\
$y_p = -\frac{1}{2}e^{2t}\ln|t| - \frac{1}{2t}te^{2t} = -\frac{1}{2}e^{2t}\ln|t| - \frac{1}{2}e^{2t}$. (Absorb $-\frac{1}{2}e^{2t}$). \\
\textbf{Final Answer:} $y(t) = c_1e^{2t} + c_2te^{2t} - \frac{1}{2}e^{2t}\ln|t|$

\subsection*{Concept 4 Solutions}
\textbf{16.} $y'' + y = \sec(t) + e^{2t}$ \\
For $\sec(t)$ (Variation of Parameters): $y_{p1} = \cos(t)\ln|\cos(t)| + t\sin(t)$. \\
For $e^{2t}$ (Undetermined Coefficients): Assume $y_{p2} = Ae^{2t}$. $4A + A = 1 \implies A = 1/5$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(t) + c_2\sin(t) + \cos(t)\ln|\cos(t)| + t\sin(t) + \frac{1}{5}e^{2t}$

\vspace{0.5cm}
\textbf{17.} $y'' - 2y' + y = \frac{e^t}{t} + 3t$ \\
For $e^t/t$ (Variation of Parameters): $y_{p1} = te^t\ln|t|$. \\
For $3t$ (Undetermined Coefficients): Assume $y_{p2} = At + B$. $0 - 2(A) + (At+B) = 3t \implies A=3, B=6$. \\
\textbf{Final Answer:} $y(t) = c_1e^t + c_2te^t + te^t\ln|t| + 3t + 6$

\vspace{0.5cm}
\textbf{18.} $y'' + 4y = \tan(2t) + 5e^{-t}$ \\
For $\tan(2t)$: $y_{p1} = -\frac{1}{4}\cos(2t)\ln|\sec(2t) + \tan(2t)|$. \\
For $5e^{-t}$: Assume $y_{p2} = Ae^{-t}$. $A + 4A = 5 \implies A = 1$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(2t) + c_2\sin(2t) - \frac{1}{4}\cos(2t)\ln|\sec(2t) + \tan(2t)| + e^{-t}$

\vspace{0.5cm}
\textbf{19.} $y'' + 9y = \csc(3t) + \sin(t)$ \\
For $\csc(3t)$: $y_{p1} = -\frac{t}{3}\cos(3t) + \frac{1}{9}\sin(3t)\ln|\sin(3t)|$. \\
For $\sin(t)$: Assume $y_{p2} = A\sin(t)$. $-A + 9A = 1 \implies A = 1/8$. \\
\textbf{Final Answer:} $y(t) = c_1\cos(3t) + c_2\sin(3t) - \frac{t}{3}\cos(3t) + \frac{1}{9}\sin(3t)\ln|\sin(3t)| + \frac{1}{8}\sin(t)$

\subsection*{Concept 5 Solutions}
\textbf{20.} $y'' - y = \frac{1}{t^2}$, $y(1) = 0$, $y'(1) = 0$ \\
$y_1 = e^t, y_2 = e^{-t}, W = -2$. Integrate from $t_0 = 1$ to $t$. \\
$v_1(t) = \int_1^t \frac{-e^{-s}(1/s^2)}{-2} \, ds = \int_1^t \frac{e^{-s}}{2s^2} \, ds$. \\
$v_2(t) = \int_1^t \frac{e^s(1/s^2)}{-2} \, ds = -\int_1^t \frac{e^s}{2s^2} \, ds$. \\
$y(t) = e^t \int_1^t \frac{e^{-s}}{2s^2} \, ds - e^{-t} \int_1^t \frac{e^s}{2s^2} \, ds = \int_1^t \frac{e^{t-s} - e^{-(t-s)}}{2s^2} \, ds$. \\
\textbf{Final Answer:} $y(t) = \int_1^t \frac{\sinh(t-s)}{s^2} \, ds$

\vspace{0.5cm}
\textbf{21.} $y'' + y = e^{-t^2}$, $y(0) = 0$, $y'(0) = 0$ \\
$y_1 = \cos(t), y_2 = \sin(t), W = 1$. Integrate from $t_0 = 0$ to $t$. \\
$v_1(t) = \int_0^t -\sin(s)e^{-s^2} \, ds$. $v_2(t) = \int_0^t \cos(s)e^{-s^2} \, ds$. \\
$y(t) = \cos(t)\int_0^t -\sin(s)e^{-s^2} \, ds + \sin(t)\int_0^t \cos(s)e^{-s^2} \, ds = \int_0^t e^{-s^2}(\sin(t)\cos(s) - \cos(t)\sin(s)) \, ds$. \\
\textbf{Final Answer:} $y(t) = \int_0^t e^{-s^2}\sin(t-s) \, ds$

\vspace{0.5cm}
\textbf{22.} Show $y_p(t) = \int_0^t f(s) \frac{\sin(2(t-s))}{2} \, ds$ for $y'' + 4y = f(t)$. \\
$y_1 = \cos(2t), y_2 = \sin(2t), W = 2$. \\
$v_1(t) = \int_0^t \frac{-\sin(2s)f(s)}{2} \, ds$, $v_2(t) = \int_0^t \frac{\cos(2s)f(s)}{2} \, ds$. \\
$y_p(t) = \int_0^t f(s) \frac{\sin(2t)\cos(2s) - \cos(2t)\sin(2s)}{2} \, ds = \int_0^t f(s) \frac{\sin(2(t-s))}{2} \, ds$. (Proved).

\vspace{0.5cm}
\textbf{23.} Show $y_p(t) = \int_0^t f(s) (t-s)e^{t-s} \, ds$ for $y'' - 2y' + y = f(t)$. \\
$y_1 = e^t, y_2 = te^t, W = e^{2t}$. \\
$v_1(t) = \int_0^t \frac{-se^s f(s)}{e^{2s}} \, ds = \int_0^t -se^{-s}f(s) \, ds$. \\
$v_2(t) = \int_0^t \frac{e^s f(s)}{e^{2s}} \, ds = \int_0^t e^{-s}f(s) \, ds$. \\
$y_p(t) = e^t \int_0^t -se^{-s}f(s) \, ds + te^t \int_0^t e^{-s}f(s) \, ds = \int_0^t f(s) e^{t-s}(t-s) \, ds$. (Proved).

\subsection*{Concept 6 Solutions}
\textbf{24.} $t^2 y'' - 2y = t^3$ \\
Standard form: $y'' - \frac{2}{t^2}y = t$. $f(t) = t$. $W = t^2(-t^{-2}) - 2t(t^{-1}) = -3$. \\
$v_1' = -\frac{t^{-1}(t)}{-3} = \frac{1}{3} \implies v_1 = \frac{t}{3}$. \\
$v_2' = \frac{t^2(t)}{-3} = -\frac{t^3}{3} \implies v_2 = -\frac{t^4}{12}$. \\
$y_p = \frac{t}{3}(t^2) - \frac{t^4}{12}(t^{-1}) = \frac{t^3}{3} - \frac{t^3}{12} = \frac{t^3}{4}$. \\
\textbf{Final Answer:} $y_p(t) = \frac{t^3}{4}$

\vspace{0.5cm}
\textbf{25.} $t^2 y'' + t y' - y = t^2$ \\
Standard form: $y'' + \frac{1}{t}y' - \frac{1}{t^2}y = 1$. $f(t) = 1$. $W = t(-t^{-2}) - 1(t^{-1}) = -\frac{2}{t}$. \\
$v_1' = -\frac{t^{-1}(1)}{-2/t} = \frac{1}{2} \implies v_1 = \frac{t}{2}$. \\
$v_2' = \frac{t(1)}{-2/t} = -\frac{t^2}{2} \implies v_2 = -\frac{t^3}{6}$. \\
$y_p = \frac{t}{2}(t) - \frac{t^3}{6}(t^{-1}) = \frac{t^2}{2} - \frac{t^2}{6} = \frac{t^2}{3}$. \\
\textbf{Final Answer:} $y_p(t) = \frac{t^2}{3}$

\vspace{0.5cm}
\textbf{26.} $t^2 y'' - 3t y' + 3y = t^4$ \\
Standard form: $y'' - \frac{3}{t}y' + \frac{3}{t^2}y = t^2$. $f(t) = t^2$. $W = t(3t^2) - 1(t^3) = 2t^3$. \\
$v_1' = -\frac{t^3(t^2)}{2t^3} = -\frac{t^2}{2} \implies v_1 = -\frac{t^3}{6}$. \\
$v_2' = \frac{t(t^2)}{2t^3} = \frac{1}{2} \implies v_2 = \frac{t}{2}$. \\
$y_p = -\frac{t^3}{6}(t) + \frac{t}{2}(t^3) = \frac{t^4}{3}$. \\
\textbf{Final Answer:} $y_p(t) = \frac{t^4}{3}$

\vspace{0.5cm}
\textbf{27.} $t^2 y'' + 2t y' - 2y = t^{-1}$ \\
Standard form: $y'' + \frac{2}{t}y' - \frac{2}{t^2}y = t^{-3}$. $f(t) = t^{-3}$. $W = t(-2t^{-3}) - 1(t^{-2}) = -3t^{-2}$. \\
$v_1' = -\frac{t^{-2}(t^{-3})}{-3t^{-2}} = \frac{t^{-3}}{3} \implies v_1 = -\frac{t^{-2}}{6}$. \\
$v_2' = \frac{t(t^{-3})}{-3t^{-2}} = -\frac{1}{3} \implies v_2 = -\frac{t}{3}$. \\
$y_p = -\frac{t^{-2}}{6}(t) - \frac{t}{3}(t^{-2}) = -\frac{t^{-1}}{6} - \frac{t^{-1}}{3} = -\frac{t^{-1}}{2}$. \\
\textbf{Final Answer:} $y_p(t) = -\frac{1}{2t}$

\subsection*{Concept 7 Solutions}
\textbf{28.} \textbf{Error Identification:} The student forgot the negative sign in the formula for $v_1'$. The formula is $v_1' = \frac{-y_2 f(t)}{W}$. \\
\textbf{Corrected Solution:} $v_1' = -\sin(t)\sec(t) = -\tan(t) \implies v_1 = \ln|\cos(t)|$. \\
$y_p = \cos(t)\ln|\cos(t)| + t\sin(t)$.

\vspace{0.5cm}
\textbf{29.} \textbf{Error Identification:} The student forgot to divide by the leading coefficient $a=3$. The formulas require dividing by $aW$. \\
\textbf{Corrected Solution:} $v_1' = -\frac{1}{3}\sin(t)\tan(t)$ and $v_2' = \frac{1}{3}\cos(t)\tan(t)$. \\
$y_p = -\frac{1}{3}\cos(t)\ln|\sec(t)+\tan(t)|$.

\vspace{0.5cm}
\textbf{30.} \textbf{Error Identification:} The student used the wrong formula for the Wronskian. They used addition instead of subtraction ($W = y_1 y_2' - y_1' y_2$). \\
\textbf{Corrected Solution:} $W = e^t(e^t + te^t) - e^t(te^t) = e^{2t}$. \\
$v_1' = \frac{-te^t(e^t/t)}{e^{2t}} = -1 \implies v_1 = -t$. \\
$v_2' = \frac{e^t(e^t/t)}{e^{2t}} = \frac{1}{t} \implies v_2 = \ln|t|$. \\
$y_p = -te^t + te^t\ln|t|$.
