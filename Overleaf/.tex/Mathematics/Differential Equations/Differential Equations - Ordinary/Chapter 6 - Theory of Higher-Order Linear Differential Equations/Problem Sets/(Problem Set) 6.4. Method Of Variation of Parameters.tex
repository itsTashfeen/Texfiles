\textbf{Directions:} For problems 26--30, act as the grader. Each solution contains a single, subtle, yet critical flaw. Find the error, explain why it is wrong in one sentence, and provide the correct final solution.
\begin{enumerate}[resume]
    \item \textbf{Solve:} $y'' - y = e^x$ \\
    \textbf{Flawed Solution:} $y_1 = e^x, y_2 = e^{-x}$. $W = -2$. $v_1' = \frac{-e^{-x}(e^x)}{-2} = \frac{1}{2} \implies v_1 = \frac{x}{2}$. $v_2' = \frac{e^x(e^x)}{-2} = -\frac{e^{2x}}{2} \implies v_2 = -\frac{e^{2x}}{4}$. $y_p = \frac{x}{2} - \frac{e^{2x}}{4}$.
    
    \item \textbf{Solve:} $x^2 y'' - 2xy' + 2y = x^3$ \\
    \textbf{Flawed Solution:} $y_1 = x, y_2 = x^2$. $W = x^2$. $g(x) = x^3$. $v_1' = \frac{-x^2(x^3)}{x^2} = -x^3 \implies v_1 = -\frac{x^4}{4}$. $v_2' = \frac{x(x^3)}{x^2} = x^2 \implies v_2 = \frac{x^3}{3}$. $y_p = (-\frac{x^4}{4})(x) + (\frac{x^3}{3})(x^2) = \frac{x^5}{12}$.
    
    \item \textbf{Solve:} $y'' + y = \tan x$ \\
    \textbf{Flawed Solution:} $y_1 = \cos x, y_2 = \sin x$. $W = 1$. $v_1' = -\sin x \tan x$. Integrating: $\int -\sin x \frac{\sin x}{\cos x} dx = \int -\cos x \tan x dx = -\sin x$.
    
    \item \textbf{Solve:} $y''' - y'' = 1$ \\
    \textbf{Flawed Solution:} $y_1 = 1, y_2 = x, y_3 = e^x$. $W = e^x$. To find $W_1$, we replace the first row of the Wronskian with $[0, 0, 1]$: $W_1 = \begin{vmatrix} 0 & 0 & 1 \\ 0 & 1 & e^x \\ 0 & 0 & e^x \end{vmatrix} = 0$.
    
    \item \textbf{Solve:} $y'' - 4y = e^{2x}$ \\
    \textbf{Flawed Solution:} $y_1 = e^{2x}, y_2 = e^{-2x}$. The Wronskian is $W = y_1 y_2 + y_1' y_2' = (e^{2x})(e^{-2x}) + (2e^{2x})(-2e^{-2x}) = 1 - 4 = -3$.
\end{enumerate}

\section{Detailed Solutions}

\subsection*{Problem 1}
\textbf{Equation:} $y'' - 4y = e^{2x}$ \\
\textbf{Step 1:} Characteristic equation $r^2 - 4 = 0 \implies r = \pm 2$. Fundamental set: $\{e^{2x}, e^{-2x}\}$. \\
\textbf{Step 2:} Wronskian $W = \begin{vmatrix} e^{2x} & e^{-2x} \\ 2e^{2x} & -2e^{-2x} \end{vmatrix} = -2 - 2 = -4$. \\
\textbf{Step 3:} $v_1' = \frac{-e^{-2x}(e^{2x})}{-4} = \frac{1}{4} \implies v_1 = \frac{1}{4}x$. \\
$v_2' = \frac{e^{2x}(e^{2x})}{-4} = -\frac{1}{4}e^{4x} \implies v_2 = -\frac{1}{16}e^{4x}$. \\
\textbf{Step 4:} $y_p = (\frac{1}{4}x)e^{2x} + (-\frac{1}{16}e^{4x})e^{-2x} = \frac{1}{4}xe^{2x} - \frac{1}{16}e^{2x}$. \\
Since $-\frac{1}{16}e^{2x}$ is a homogeneous solution, we drop it. \\
\textbf{Final Answer:} $y_p(x) = \frac{1}{4}xe^{2x}$

\subsection*{Problem 2}
\textbf{Equation:} $y'' + 9y = \sec(3x)$ \\
\textbf{Step 1:} $r^2 + 9 = 0 \implies r = \pm 3i$. Fundamental set: $\{\cos(3x), \sin(3x)\}$. \\
\textbf{Step 2:} $W = \begin{vmatrix} \cos(3x) & \sin(3x) \\ -3\sin(3x) & 3\cos(3x) \end{vmatrix} = 3(\cos^2(3x) + \sin^2(3x)) = 3$. \\
\textbf{Step 3:} $v_1' = \frac{-\sin(3x)\sec(3x)}{3} = -\frac{1}{3}\tan(3x) \implies v_1 = \frac{1}{9}\ln|\cos(3x)|$. \\
$v_2' = \frac{\cos(3x)\sec(3x)}{3} = \frac{1}{3} \implies v_2 = \frac{1}{3}x$. \\
\textbf{Final Answer:} $y_p(x) = \frac{1}{9}\cos(3x)\ln|\cos(3x)| + \frac{1}{3}x\sin(3x)$

\subsection*{Problem 3}
\textbf{Equation:} $y'' - 2y' + y = \frac{e^x}{x}$ \\
\textbf{Step 1:} $r^2 - 2r + 1 = 0 \implies r = 1$ (multiplicity 2). Fundamental set: $\{e^x, xe^x\}$. \\
\textbf{Step 2:} $W = \begin{vmatrix} e^x & xe^x \\ e^x & (x+1)e^x \end{vmatrix} = e^{2x}(x+1) - xe^{2x} = e^{2x}$. \\
\textbf{Step 3:} $v_1' = \frac{-xe^x(e^x/x)}{e^{2x}} = -1 \implies v_1 = -x$. \\
$v_2' = \frac{e^x(e^x/x)}{e^{2x}} = \frac{1}{x} \implies v_2 = \ln|x|$. \\
\textbf{Step 4:} $y_p = -xe^x + xe^x\ln|x|$. The term $-xe^x$ is homogeneous, so we drop it. \\
\textbf{Final Answer:} $y_p(x) = xe^x\ln|x|$

\subsection*{Problem 4}
\textbf{Equation:} $y'' - y = \cosh x$ \\
\textbf{Step 1:} $r^2 - 1 = 0 \implies r = \pm 1$. Fundamental set: $\{e^x, e^{-x}\}$. Note: $\cosh x = \frac{e^x + e^{-x}}{2}$. \\
\textbf{Step 2:} $W = \begin{vmatrix} e^x & e^{-x} \\ e^x & -e^{-x} \end{vmatrix} = -2$. \\
\textbf{Step 3:} $v_1' = \frac{-e^{-x}(\frac{e^x + e^{-x}}{2})}{-2} = \frac{1 + e^{-2x}}{4} \implies v_1 = \frac{x}{4} - \frac{e^{-2x}}{8}$. \\
$v_2' = \frac{e^x(\frac{e^x + e^{-x}}{2})}{-2} = -\frac{e^{2x} + 1}{4} \implies v_2 = -\frac{e^{2x}}{8} - \frac{x}{4}$. \\
\textbf{Step 4:} $y_p = (\frac{x}{4} - \frac{e^{-2x}}{8})e^x + (-\frac{e^{2x}}{8} - \frac{x}{4})e^{-x} = \frac{x}{4}e^x - \frac{1}{8}e^{-x} - \frac{1}{8}e^x - \frac{x}{4}e^{-x}$. \\
Dropping homogeneous terms ($-\frac{1}{8}e^{-x}$ and $-\frac{1}{8}e^x$), we get $\frac{x}{4}e^x - \frac{x}{4}e^{-x} = \frac{x}{2}\sinh x$. \\
\textbf{Final Answer:} $y_p(x) = \frac{x}{2}\sinh x$

\subsection*{Problem 5}
\textbf{Equation:} $y''' - y'' = 2x$ \\
\textbf{Step 1:} $r^3 - r^2 = 0 \implies r = 0, 0, 1$. Fundamental set: $\{1, x, e^x\}$. \\
\textbf{Step 2:} $W = \begin{vmatrix} 1 & x & e^x \\ 0 & 1 & e^x \\ 0 & 0 & e^x \end{vmatrix} = e^x$. \\
\textbf{Step 3:} $W_1 = \begin{vmatrix} 0 & x & e^x \\ 0 & 1 & e^x \\ 1 & 0 & e^x \end{vmatrix} = e^x(x-1)$. $W_2 = \begin{vmatrix} 1 & 0 & e^x \\ 0 & 0 & e^x \\ 0 & 1 & e^x \end{vmatrix} = -e^x$. $W_3 = \begin{vmatrix} 1 & x & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1 \end{vmatrix} = 1$. \\
\textbf{Step 4:} $v_1' = \frac{2x \cdot e^x(x-1)}{e^x} = 2x^2 - 2x \implies v_1 = \frac{2}{3}x^3 - x^2$. \\
$v_2' = \frac{2x(-e^x)}{e^x} = -2x \implies v_2 = -x^2$. \\
$v_3' = \frac{2x(1)}{e^x} = 2xe^{-x} \implies v_3 = -2xe^{-x} - 2e^{-x}$. \\
\textbf{Step 5:} $y_p = (\frac{2}{3}x^3 - x^2)(1) + (-x^2)(x) + (-2xe^{-x} - 2e^{-x})e^x = \frac{2}{3}x^3 - x^2 - x^3 - 2x - 2$. \\
Drop the constant $-2$ (homogeneous). \\
\textbf{Final Answer:} $y_p(x) = -\frac{1}{3}x^3 - x^2 - 2x$

\subsection*{Problem 6}
\textbf{Equation:} $y''' + y' = \tan x$ \\
\textbf{Step 1:} $r^3 + r = 0 \implies r = 0, \pm i$. Fundamental set: $\{1, \cos x, \sin x\}$. \\
\textbf{Step 2:} $W = \begin{vmatrix} 1 & \cos x & \sin x \\ 0 & -\sin x & \cos x \\ 0 & -\cos x & -\sin x \end{vmatrix} = 1(\sin^2 x + \cos^2 x) = 1$. \\
\textbf{Step 3:} $W_1 = 1$, $W_2 = -\cos x$, $W_3 = -\sin x$. \\
\textbf{Step 4:} $v_1' = \tan x \implies v_1 = -\ln|\cos x|$. \\
$v_2' = -\sin x \implies v_2 = \cos x$. \\
$v_3' = -\frac{\sin^2 x}{\cos x} = \cos x - \sec x \implies v_3 = \sin x - \ln|\sec x + \tan x|$. \\
\textbf{Step 5:} $y_p = -\ln|\cos x| + \cos^2 x + \sin^2 x - \sin x\ln|\sec x + \tan x|$. Drop $\cos^2 x + \sin^2 x = 1$. \\
\textbf{Final Answer:} $y_p(x) = -\ln|\cos x| - \sin x\ln|\sec x + \tan x|$

\subsection*{Problem 7}
\textbf{Equation:} $y''' - 2y'' - y' + 2y = e^{3x}$ \\
\textbf{Step 1:} $r^3 - 2r^2 - r + 2 = 0 \implies r = 1, -1, 2$. Fundamental set: $\{e^x, e^{-x}, e^{2x}\}$. \\
\textbf{Step 2:} $W = -6e^{2x}$. $W_1 = 3e^x$, $W_2 = -e^{3x}$, $W_3 = -2$. \\
\textbf{Step 3:} $v_1' = \frac{e^{3x}(3e^x)}{-6e^{2x}} = -\frac{1}{2}e^{2x} \implies v_1 = -\frac{1}{4}e^{2x}$. \\
$v_2' = \frac{e^{3x}(-e^{3x})}{-6e^{2x}} = \frac{1}{6}e^{4x} \implies v_2 = \frac{1}{24}e^{4x}$. \\
$v_3' = \frac{e^{3x}(-2)}{-6e^{2x}} = \frac{1}{3}e^x \implies v_3 = \frac{1}{3}e^x$. \\
\textbf{Step 4:} $y_p = -\frac{1}{4}e^{3x} + \frac{1}{24}e^{3x} + \frac{1}{3}e^{3x} = \frac{-6+1+8}{24}e^{3x} = \frac{3}{24}e^{3x}$. \\
\textbf{Final Answer:} $y_p(x) = \frac{1}{8}e^{3x}$

\subsection*{Problem 8}
\textbf{Equation:} $y''' - 3y'' + 3y' - y = \frac{e^x}{x}$ \\
\textbf{Step 1:} $(r-1)^3 = 0 \implies r = 1, 1, 1$. Fundamental set: $\{e^x, xe^x, x^2e^x\}$. \\
\textbf{Step 2:} $W = 2e^{3x}$. $W_1 = x^2e^{2x}$, $W_2 = -2xe^{2x}$, $W_3 = e^{2x}$. \\
\textbf{Step 3:} $v_1' = \frac{(e^x/x)(x^2e^{2x})}{2e^{3x}} = \frac{x}{2} \implies v_1 = \frac{x^2}{4}$. \\
$v_2' = \frac{(e^x/x)(-2xe^{2x})}{2e^{3x}} = -1 \implies v_2 = -x$. \\
$v_3' = \frac{(e^x/x)(e^{2x})}{2e^{3x}} = \frac{1}{2x} \implies v_3 = \frac{1}{2}\ln|x|$. \\
\textbf{Step 4:} $y_p = \frac{x^2}{4}e^x - x^2e^x + \frac{1}{2}x^2e^x\ln|x|$. Drop $-\frac{3}{4}x^2e^x$ (homogeneous). \\
\textbf{Final Answer:} $y_p(x) = \frac{1}{2}x^2e^x\ln|x|$

\subsection*{Problem 9}
\textbf{Equation:} $y'' + 4y' + 4y = x^{-2}e^{-2x}$ \\
\textbf{Step 1:} $(r+2)^2 = 0 \implies r = -2, -2$. Fundamental set: $\{e^{-2x}, xe^{-2x}\}$. \\
\textbf{Step 2:} $W = e^{-4x}$. \\
\textbf{Step 3:} $v_1' = \frac{-xe^{-2x}(x^{-2}e^{-2x})}{e^{-4x}} = -\frac{1}{x} \implies v_1 = -\ln|x|$. \\
$v_2' = \frac{e^{-2x}(x^{-2}e^{-2x})}{e^{-4x}} = \frac{1}{x^2} \implies v_2 = -\frac{1}{x}$. \\
\textbf{Step 4:} $y_p = -e^{-2x}\ln|x| - e^{-2x}$. Drop $-e^{-2x}$. \\
\textbf{Final Answer:} $y_p(x) = -e^{-2x}\ln|x|$

\subsection*{Problem 10}
\textbf{Equation:} $y''' + 3y'' + 3y' + y = e^{-x} \ln x$ \\
\textbf{Step 1:} $(r+1)^3 = 0 \implies r = -1, -1, -1$. Fundamental set: $\{e^{-x}, xe^{-x}, x^2e^{-x}\}$. \\
\textbf{Step 2:} $W = 2e^{-3x}$. $W_1 = x^2e^{-2x}$, $W_2 = -2xe^{-2x}$, $W_3 = e^{-2x}$. \\
\textbf{Step 3:} $v_1' = \frac{1}{2}x^2\ln x \implies v_1 = \frac{1}{6}x^3\ln x - \frac{1}{18}x^3$. \\
$v_2' = -x\ln x \implies v_2 = -\frac{1}{2}x^2\ln x + \frac{1}{4}x^2$. \\
$v_3' = \frac{1}{2}\ln x \implies v_3 = \frac{1}{2}x\ln x - \frac{1}{2}x$. \\
\textbf{Step 4:} $y_p = (\frac{1}{6} - \frac{1}{2} + \frac{1}{2})x^3e^{-x}\ln x + (-\frac{1}{18} + \frac{1}{4} - \frac{1}{2})x^3e^{-x}$. \\
\textbf{Final Answer:} $y_p(x) = \frac{1}{6}x^3e^{-x}\ln x - \frac{11}{36}x^3e^{-x}$

\subsection*{Problem 11}
\textbf{Equation:} $y'' - 6y' + 9y = \frac{e^{3x}}{x^2}$ \\
\textbf{Step 1:} $(r-3)^2 = 0 \implies r = 3, 3$. Fundamental set: $\{e^{3x}, xe^{3x}\}$. \\
\textbf{Step 2:} $W = e^{6x}$. \\
\textbf{Step 3:} $v_1' = \frac{-xe^{3x}(e^{3x}/x^2)}{e^{6x}} = -\frac{1}{x} \implies v_1 = -\ln|x|$. \\
$v_2' = \frac{e^{3x}(e^{3x}/x^2)}{e^{6x}} = \frac{1}{x^2} \implies v_2 = -\frac{1}{x}$. \\
\textbf{Step 4:} $y_p = -e^{3x}\ln|x| - e^{3x}$. Drop $-e^{3x}$. \\
\textbf{Final Answer:} $y_p(x) = -e^{3x}\ln|x|$

\subsection*{Problem 12}
\textbf{Equation:} $y'' + y = \csc x$ \\
\textbf{Step 1:} $r^2 + 1 = 0 \implies r = \pm i$. Fundamental set: $\{\cos x, \sin x\}$. \\
\textbf{Step 2:} $W = 1$. \\
\textbf{Step 3:} $v_1' = -\sin x \csc x = -1 \implies v_1 = -x$. \\
$v_2' = \cos x \csc x = \cot x \implies v_2 = \ln|\sin x|$. \\
\textbf{Final Answer:} $y_p(x) = -x\cos x + \sin x\ln|\sin x|$

\subsection*{Problem 13}
\textbf{Equation:} $y'' + 4y = \tan(2x)$ \\
\textbf{Step 1:} $r^2 + 4 = 0 \implies r = \pm 2i$. Fundamental set: $\{\cos(2x), \sin(2x)\}$. \\
\textbf{Step 2:} $W = 2$. \\
\textbf{Step 3:} $v_1' = -\frac{1}{2}\sin(2x)\tan(2x) = \frac{1}{2}(\cos(2x) - \sec(2x)) \implies v_1 = \frac{1}{4}\sin(2x) - \frac{1}{4}\ln|\sec(2x) + \tan(2x)|$. \\
$v_2' = \frac{1}{2}\cos(2x)\tan(2x) = \frac{1}{2}\sin(2x) \implies v_2 = -\frac{1}{4}\cos(2x)$. \\
\textbf{Step 4:} $y_p = \frac{1}{4}\sin(2x)\cos(2x) - \frac{1}{4}\cos(2x)\ln|\sec(2x) + \tan(2x)| - \frac{1}{4}\sin(2x)\cos(2x)$. \\
\textbf{Final Answer:} $y_p(x) = -\frac{1}{4}\cos(2x)\ln|\sec(2x) + \tan(2x)|$

\subsection*{Problem 14}
\textbf{Equation:} $y'' - 2y' + 2y = e^x \sec x$ \\
\textbf{Step 1:} $r^2 - 2r + 2 = 0 \implies r = 1 \pm i$. Fundamental set: $\{e^x\cos x, e^x\sin x\}$. \\
\textbf{Step 2:} $W = e^{2x}$. \\
\textbf{Step 3:} $v_1' = \frac{-e^x\sin x(e^x\sec x)}{e^{2x}} = -\tan x \implies v_1 = \ln|\cos x|$. \\
$v_2' = \frac{e^x\cos x(e^x\sec x)}{e^{2x}} = 1 \implies v_2 = x$. \\
\textbf{Final Answer:} $y_p(x) = e^x\cos x\ln|\cos x| + xe^x\sin x$

\subsection*{Problem 15}
\textbf{Equation:} $y'' + y = \sec^2 x$ \\
\textbf{Step 1:} $r^2 + 1 = 0 \implies r = \pm i$. Fundamental set: $\{\cos x, \sin x\}$. \\
\textbf{Step 2:} $W = 1$. \\
\textbf{Step 3:} $v_1' = -\sin x \sec^2 x = -\tan x \sec x \implies v_1 = -\sec x$. \\
$v_2' = \cos x \sec^2 x = \sec x \implies v_2 = \ln|\sec x + \tan x|$. \\
\textbf{Step 4:} $y_p = -\sec x \cos x + \sin x \ln|\sec x + \tan x| = -1 + \sin x \ln|\sec x + \tan x|$. \\
\textbf{Final Answer:} $y_p(x) = -1 + \sin x \ln|\sec x + \tan x|$

\subsection*{Problem 16}
\textbf{Equation:} $x^2 y'' - 2y = 3x^2 - 1$ \\
\textbf{Step 1:} Cauchy-Euler. $r(r-1) - 2 = 0 \implies r^2 - r - 2 = 0 \implies r = 2, -1$. Fundamental set: $\{x^2, x^{-1}\}$. \\
\textbf{Step 2:} Standard form: $y'' - \frac{2}{x^2}y = 3 - x^{-2}$. Thus $g(x) = 3 - x^{-2}$. \\
\textbf{Step 3:} $W = \begin{vmatrix} x^2 & x^{-1} \\ 2x & -x^{-2} \end{vmatrix} = -1 - 2 = -3$. \\
\textbf{Step 4:} $v_1' = \frac{-x^{-1}(3 - x^{-2})}{-3} = x^{-1} - \frac{1}{3}x^{-3} \implies v_1 = \ln|x| + \frac{1}{6}x^{-2}$. \\
$v_2' = \frac{x^2(3 - x^{-2})}{-3} = -x^2 + \frac{1}{3} \implies v_2 = -\frac{1}{3}x^3 + \frac{1}{3}x$. \\
\textbf{Step 5:} $y_p = x^2\ln|x| + \frac{1}{6} - \frac{1}{3}x^2 + \frac{1}{3}$. Drop $-\frac{1}{3}x^2$. \\

\subsection*{Problem 17}
\textbf{Equation:} $x^2 y'' + xy' - y = x \ln x$ \\

\subsection*{Problem 25}
\textbf{Derivation:} \\
Assume a particular solution of the form $y_p = v_1y_1 + v_2y_2$. \\
Differentiating yields $y_p' = (v_1y_1' + v_2y_2') + (v_1'y_1 + v_2'y_2)$. \\
To prevent second derivatives of $v_k$ from appearing, we impose the first condition: $v_1'y_1 + v_2'y_2 = 0$. \\
This simplifies the first derivative to $y_p' = v_1y_1' + v_2y_2'$. \\
Differentiating again yields $y_p'' = v_1y_1'' + v_2y_2'' + v_1'y_1' + v_2'y_2'$. \\
Substitute $y_p, y_p', y_p''$ into the original ODE $y'' + p(x)y' + q(x)y = g(x)$: \\
$(v_1y_1'' + v_2y_2'' + v_1'y_1' + v_2'y_2') + p(x)(v_1y_1' + v_2y_2') + q(x)(v_1y_1 + v_2y_2) = g(x)$. \\
Rearranging by factoring out $v_1$ and $v_2$: \\
$v_1(y_1'' + p(x)y_1' + q(x)y_1) + v_2(y_2'' + p(x)y_2' + q(x)y_2) + v_1'y_1' + v_2'y_2' = g(x)$. \\
Since $y_1$ and $y_2$ are solutions to the homogeneous equation, the terms in the parentheses are exactly zero. \\
This leaves the second condition: $v_1'y_1' + v_2'y_2' = g(x)$. $\blacksquare$
