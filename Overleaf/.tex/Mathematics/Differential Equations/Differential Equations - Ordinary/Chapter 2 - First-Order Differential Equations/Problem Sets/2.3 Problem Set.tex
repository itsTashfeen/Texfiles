\section{Detailed Solutions}

\textbf{Problem 1:} $\frac{dx}{dt} - 3xt = 5e^{2t}$ \\
\textbf{Solution:} Dependent variable is $x$, independent is $t$. Both $x$ and $dx/dt$ are to the first power, so it is \textbf{linear}. We attempt to isolate $dx/dt = 3xt + 5e^{2t}$. We cannot factor this into $g(t)h(x)$, so it is not separable. \\
\textbf{Final Answer:} Linear only.

\vspace{10pt}
\textbf{Problem 2:} $t \frac{dy}{dt} = 3y + t^2$ \\
\textbf{Solution:} Dependent variable $y$, independent $t$. Rewriting yields $y' - \frac{3}{t}y = t$, which fits the standard linear form. Isolating $dy/dt = \frac{3y+t^2}{t}$, we cannot factor this into purely $y$ and purely $t$ components. Not separable. \\
\textbf{Final Answer:} Linear only.

\vspace{10pt}
\textbf{Problem 3:} $\frac{dy}{dx} = x^2 y^3$ \\
\textbf{Solution:} The dependent variable $y$ is raised to the third power. This violates linearity. We can, however, easily separate the variables: $y^{-3} dy = x^2 dx$. \\
\textbf{Final Answer:} Separable only.

\vspace{10pt}
\textbf{Problem 4:} $\frac{dy}{dx} = \frac{y}{x}$ \\
\textbf{Solution:} We can isolate the derivative and factor it as $\frac{dy}{dx} = (\frac{1}{x})(y)$, making it separable ($\frac{1}{y}dy = \frac{1}{x}dx$). We can also write it as $y' - \frac{1}{x}y = 0$, which is standard linear form. \\
\textbf{Final Answer:} Both separable and linear.

\vspace{10pt}
\textbf{Problem 5:} $y' + 3y = 6$ \\
\textbf{Solution:} 
1. Standard form: $P(x) = 3$. 
2. $\mu(x) = e^{\int 3 dx} = e^{3x}$. 
3. Multiply: $e^{3x}y' + 3e^{3x}y = 6e^{3x} \implies (ye^{3x})' = 6e^{3x}$. 
4. Integrate: $ye^{3x} = 2e^{3x} + C$.
5. Isolate $y$: $y = 2 + Ce^{-3x}$. \\
\textbf{Final Answer:} $y = 2 + Ce^{-3x}$

\vspace{10pt}
\textbf{Problem 6:} $x y' - y = x^2 \sin x$ \\
\textbf{Solution:} 
1. Divide by $x$: $y' - \frac{1}{x}y = x \sin x$. Here $P(x) = -1/x$. 
2. $\mu(x) = e^{\int -\frac{1}{x} dx} = e^{-\ln x} = x^{-1}$. 
3. Multiply: $x^{-1}y' - x^{-2}y = \sin x \implies (x^{-1}y)' = \sin x$. 
4. Integrate: $x^{-1}y = -\cos x + C$.
5. Isolate $y$: $y = -x\cos x + Cx$. \\
\textbf{Final Answer:} $y = -x\cos x + Cx$

\vspace{10pt}
\textbf{Problem 7:} $\frac{dy}{dx} - 2xy = e^{x^2}$ \\
\textbf{Solution:} 
1. Standard form is set. $P(x) = -2x$. 
2. $\mu(x) = e^{\int -2x dx} = e^{-x^2}$. 
3. Multiply: $e^{-x^2}y' - 2xe^{-x^2}y = 1 \implies (ye^{-x^2})' = 1$. 
4. Integrate: $ye^{-x^2} = x + C$. 
5. Isolate $y$: $y = (x+C)e^{x^2}$. \\
\textbf{Final Answer:} $y = (x+C)e^{x^2}$

\vspace{10pt}
\textbf{Problem 8:} $t \frac{dy}{dt} + 2y = t^3$ \\
\textbf{Solution:} 
1. Divide by $t$: $y' + \frac{2}{t}y = t^2$. $P(t) = 2/t$. 
2. $\mu(t) = e^{\int \frac{2}{t} dt} = e^{2\ln t} = t^2$. 
3. Multiply: $t^2y' + 2ty = t^4 \implies (t^2 y)' = t^4$. 
4. Integrate: $t^2 y = \frac{1}{5}t^5 + C$. 
5. Isolate $y$: $y = \frac{1}{5}t^3 + Ct^{-2}$. \\
\textbf{Final Answer:} $y = \frac{1}{5}t^3 + Ct^{-2}$

\vspace{10pt}
\textbf{Problem 9:} $\frac{dy}{d\theta} + (\tan \theta)y = \cos^2 \theta$ \\
\textbf{Solution:} 
1. $P(\theta) = \tan \theta = \frac{\sin \theta}{\cos \theta}$. 
2. $\mu(\theta) = e^{\int \tan \theta d\theta} = e^{-\ln|\cos \theta|} = \sec \theta$. 
3. Multiply: $\sec \theta y' + \sec \theta \tan \theta y = \cos \theta \implies (y \sec \theta)' = \cos \theta$. 
4. Integrate: $y \sec \theta = \sin \theta + C$. 
5. Isolate $y$: $y = \sin \theta \cos \theta + C\cos \theta$. \\
\textbf{Final Answer:} $y = \sin \theta \cos \theta + C\cos \theta$

\vspace{10pt}
\textbf{Problem 10:} $\frac{dy}{dx} + \frac{2}{x}y = \frac{\ln x}{x^2}$ \\
\textbf{Solution:} 
1. $P(x) = \frac{2}{x}$. 
2. $\mu(x) = e^{2\ln x} = x^2$. 
3. Multiply: $x^2 y' + 2x y = \ln x \implies (x^2 y)' = \ln x$. 
4. Integrate both sides. Use Integration by Parts for $\int \ln x \, dx$ (Let $u=\ln x, dv=dx \implies du=\frac{1}{x}dx, v=x$). $\int \ln x dx = x\ln x - x + C$. 
5. $x^2 y = x\ln x - x + C \implies y = \frac{\ln x}{x} - \frac{1}{x} + \frac{C}{x^2}$. \\
\textbf{Final Answer:} $y = \frac{\ln x}{x} - \frac{1}{x} + \frac{C}{x^2}$

\vspace{10pt}
\textbf{Problem 11:} $x y' + 4y = x^4$ \\
\textbf{Solution:} 
1. Divide by $x$: $y' + \frac{4}{x}y = x^3$. $P(x) = 4/x$. 
2. $\mu(x) = e^{4\ln x} = x^4$. 
3. Multiply: $x^4 y' + 4x^3 y = x^7 \implies (y x^4)' = x^7$. 
4. Integrate: $y x^4 = \frac{1}{8}x^8 + C$. 
5. Isolate $y$: $y = \frac{1}{8}x^4 + Cx^{-4}$. \\
\textbf{Final Answer:} $y = \frac{1}{8}x^4 + Cx^{-4}$

\vspace{10pt}
\textbf{Problem 12:} $y' - \frac{3}{x}y = x^4 e^x$ \\
\textbf{Solution:} 
1. $P(x) = -3/x$. 
2. $\mu(x) = e^{-3\ln x} = x^{-3}$. 
3. Multiply: $x^{-3}y' - 3x^{-4}y = x e^x \implies (yx^{-3})' = x e^x$. 
4. Integrate. Use Parts: $\int x e^x dx = x e^x - e^x + C$. 
5. $y x^{-3} = x e^x - e^x + C \implies y = x^4 e^x - x^3 e^x + C x^3$. \\
\textbf{Final Answer:} $y = x^4 e^x - x^3 e^x + C x^3$

\vspace{10pt}
\textbf{Problem 13:} $y' + y = x$ \\
\textbf{Solution:} 
1. $P(x) = 1$. $\mu(x) = e^x$. 
2. Multiply: $e^x y' + e^x y = x e^x \implies (ye^x)' = xe^x$. 
3. Integrate. Use Parts: $\int xe^x dx = xe^x - e^x + C$. 
4. $y e^x = xe^x - e^x + C \implies y = x - 1 + Ce^{-x}$. \\
\textbf{Final Answer:} $y = x - 1 + Ce^{-x}$

\vspace{10pt}
\textbf{Problem 14:} $t y' - y = t^2 e^{-t}$ \\
\textbf{Solution:} 
1. Divide by $t$: $y' - \frac{1}{t}y = t e^{-t}$. $P(t) = -1/t$. 
2. $\mu(t) = e^{-\ln t} = t^{-1}$. 
3. Multiply: $t^{-1}y' - t^{-2}y = e^{-t} \implies (y t^{-1})' = e^{-t}$. 
4. Integrate: $y t^{-1} = -e^{-t} + C$. 
5. Isolate $y$: $y = -te^{-t} + Ct$. \\
\textbf{Final Answer:} $y = -te^{-t} + Ct$

\vspace{10pt}
\textbf{Problem 15:} $y' + 2xy = x^3$ \\
\textbf{Solution:} 
1. $\mu(x) = e^{\int 2x dx} = e^{x^2}$. 
2. Multiply: $e^{x^2}y' + 2xe^{x^2}y = x^3 e^{x^2} \implies (ye^{x^2})' = x^3 e^{x^2}$. 
3. Integrate. $\int x^3 e^{x^2} dx = \int x^2 (x e^{x^2}) dx$. Use Parts: Let $u=x^2, dv = x e^{x^2} dx \implies du = 2x dx, v = \frac{1}{2}e^{x^2}$. 
   $uv - \int v du = \frac{1}{2}x^2 e^{x^2} - \int x e^{x^2} dx = \frac{1}{2}x^2 e^{x^2} - \frac{1}{2}e^{x^2} + C$. 
4. $y e^{x^2} = \frac{1}{2}e^{x^2}(x^2 - 1) + C \implies y = \frac{1}{2}(x^2 - 1) + Ce^{-x^2}$. \\
\textbf{Final Answer:} $y = \frac{1}{2}(x^2 - 1) + Ce^{-x^2}$

\vspace{10pt}
\textbf{Problem 16:} $y' + y = e^x, \quad y(0) = 1$ \\
\textbf{Solution:} 
1. $\mu(x) = e^x$. 
2. Multiply: $(ye^x)' = e^{2x}$. 
3. Integrate: $ye^x = \frac{1}{2}e^{2x} + C$. 
4. Isolate $y$: $y = \frac{1}{2}e^x + Ce^{-x}$. 
5. Apply IVP: $1 = \frac{1}{2}e^0 + C e^0 \implies C = \frac{1}{2}$. \\
\textbf{Final Answer:} $y = \frac{1}{2}e^x + \frac{1}{2}e^{-x}$

\vspace{10pt}
\textbf{Problem 17:} $x y' + 2y = 4x^2, \quad y(1) = 2$ \\
\textbf{Solution:} 
1. Divide by $x$: $y' + \frac{2}{x}y = 4x$. $\mu(x) = x^2$. 
2. Multiply: $(x^2 y)' = 4x^3$. 
3. Integrate: $x^2 y = x^4 + C \implies y = x^2 + Cx^{-2}$. 
4. Apply IVP: $2 = (1)^2 + C(1)^{-2} \implies C = 1$. \\
\textbf{Final Answer:} $y = x^2 + x^{-2}$

\vspace{10pt}
\textbf{Problem 18:} $y' = \frac{y}{x} + 2x + 1, \quad y(1) = 5$ \\
\textbf{Solution:} 
1. Standard form: $y' - \frac{1}{x}y = 2x + 1$. $\mu(x) = e^{-\ln x} = x^{-1}$. 
2. Multiply: $(yx^{-1})' = 2 + x^{-1}$. 
3. Integrate: $yx^{-1} = 2x + \ln|x| + C \implies y = 2x^2 + x\ln|x| + Cx$. 
4. Apply IVP: $5 = 2(1) + 0 + C \implies C = 3$. \\
\textbf{Final Answer:} $y = 2x^2 + x\ln x + 3x$

\vspace{10pt}
\textbf{Problem 19:} $\cos(x) y' + \sin(x)y = 2\cos^3(x), \quad y(0) = 1$ \\
\textbf{Solution:} 
1. Divide by $\cos(x)$: $y' + \tan(x)y = 2\cos^2(x)$. $\mu(x) = e^{\int \tan x dx} = \sec x$. 
2. Multiply: $(y \sec x)' = 2\cos x$. 
3. Integrate: $y \sec x = 2\sin x + C \implies y = 2\sin x \cos x + C\cos x$. 
4. Apply IVP: $1 = 0 + C(1) \implies C = 1$. \\
\textbf{Final Answer:} $y = 2\sin x \cos x + \cos x$

\vspace{10pt}
\textbf{Problem 20:} $\frac{dy}{dx} = \frac{1}{e^y - x}$ \\
\textbf{Solution:} 
1. Flip the derivative: $\frac{dx}{dy} = e^y - x$. 
2. Standard form in $x$: $x' + x = e^y$. Here $P(y) = 1$, so $\mu(y) = e^y$. 
3. Multiply: $(x e^y)' = e^{2y}$. 
4. Integrate wrt $y$: $x e^y = \frac{1}{2}e^{2y} + C$. 
5. Isolate $x$: $x(y) = \frac{1}{2}e^y + Ce^{-y}$. \\
\textbf{Final Answer:} $x(y) = \frac{1}{2}e^y + Ce^{-y}$

\vspace{10pt}
\textbf{Problem 21:} $y \frac{dx}{dy} + 2x = 5y^3$ \\
\textbf{Solution:} 
1. Divide by $y$: $x' + \frac{2}{y}x = 5y^2$. $\mu(y) = y^2$. 
2. Multiply: $(xy^2)' = 5y^4$. 
3. Integrate wrt $y$: $xy^2 = y^5 + C$. 
4. Isolate $x$: $x(y) = y^3 + Cy^{-2}$. \\
\textbf{Final Answer:} $x(y) = y^3 + Cy^{-2}$

\vspace{10pt}
\textbf{Problem 22:} $t^3 \frac{dy}{dt} + 3t^2 y = \cos t, \quad y(\pi) = 0$ \\
\textbf{Solution:} 
1. Notice the left side is exactly the derivative of $(t^3 y)$. 
2. Collapse: $\frac{d}{dt}[t^3 y] = \cos t$. 
3. Integrate: $t^3 y = \sin t + C$. 
4. Apply IVP: $\pi^3(0) = \sin(\pi) + C \implies 0 = 0 + C \implies C = 0$. 
5. $y = \frac{\sin t}{t^3}$. \\
\textbf{Final Answer:} $y = \frac{\sin t}{t^3}$

\vspace{10pt}
\textbf{Problem 23:} $e^{2t} \frac{dx}{dt} + 2e^{2t} x = t, \quad x(0) = 1$ \\
\textbf{Solution:} 
1. The left side is pre-collapsed: $\frac{d}{dt}[e^{2t} x] = t$. 
2. Integrate: $e^{2t} x = \frac{1}{2}t^2 + C$. 
3. Apply IVP: $e^0 (1) = 0 + C \implies C = 1$. 
4. $x(t) = (\frac{1}{2}t^2 + 1)e^{-2t}$. \\
\textbf{Final Answer:} $x(t) = (\frac{1}{2}t^2 + 1)e^{-2t}$

\vspace{10pt}
\textbf{Problem 24:} $y' + y = y^2 e^x$ \\
\textbf{Solution:} 
1. Bernoulli with $n=2$. Let $v = y^{1-2} = y^{-1}$. Then $v' = -y^{-2}y'$. 
2. Divide original by $y^2$: $y^{-2}y' + y^{-1} = e^x$. 
3. Substitute: $-v' + v = e^x \implies v' - v = -e^x$. 
4. Linear in $v$. $\mu(x) = e^{-x}$. 
5. Multiply: $(ve^{-x})' = -e^x \cdot e^{-x} = -1$. 
6. Integrate: $v e^{-x} = -x + C \implies v = (C - x)e^x$. 
7. Reverse sub: $y = v^{-1} = \frac{1}{(C - x)e^x}$. \\
\textbf{Final Answer:} $y = \frac{1}{(C - x)e^x}$

\vspace{10pt}
\textbf{Problem 25:} $x y' + y = x^2 y^3$ \\
\textbf{Solution:} 
1. Divide by $x$: $y' + \frac{1}{x}y = x y^3$. $n=3$. Let $v = y^{-2}$, $v' = -2y^{-3}y'$. 
2. Multiply by $-2y^{-3}$: $-2y^{-3}y' - \frac{2}{x}y^{-2} = -2x \implies v' - \frac{2}{x}v = -2x$. 
3. $\mu(x) = e^{-2\ln x} = x^{-2}$. 
4. Multiply: $(vx^{-2})' = -2x \cdot x^{-2} = -2x^{-1}$. 
5. Integrate: $vx^{-2} = -2\ln|x| + C \implies v = x^2(C - 2\ln|x|)$. 
6. Reverse sub: $y^{-2} = x^2(C - 2\ln|x|) \implies y = \pm \frac{1}{x\sqrt{C - 2\ln|x|}}$. \\
\textbf{Final Answer:} $y = \pm \frac{1}{x\sqrt{C - 2\ln|x|}}$

\vspace{10pt}
\textbf{Problem 26:} $y' - \frac{1}{x}y = -\frac{1}{2y}$ \\
\textbf{Solution:} 
1. Rewrite RHS: $-\frac{1}{2}y^{-1}$. Bernoulli with $n=-1$. Let $v = y^2, v' = 2yy'$. 
2. Multiply by $2y$: $2yy' - \frac{2}{x}y^2 = -1 \implies v' - \frac{2}{x}v = -1$. 
3. $\mu(x) = x^{-2}$. 
4. Multiply: $(v x^{-2})' = -x^{-2}$. 
5. Integrate: $v x^{-2} = x^{-1} + C \implies v = x + Cx^2$. 
6. Reverse sub: $y^2 = x + Cx^2 \implies y = \pm\sqrt{x + Cx^2}$. \\
\textbf{Final Answer:} $y = \pm\sqrt{x + Cx^2}$

\vspace{10pt}
\textbf{Problem 27:} Continuous Annuity \\
\textbf{Solution:} 
1. Model setup: $V' = rV + D(t) \implies V' - 0.04V = 4000e^{0.02t}$. 
2. $\mu(t) = e^{-0.04t}$. 
3. Multiply: $(Ve^{-0.04t})' = 4000e^{0.02t}e^{-0.04t} = 4000e^{-0.02t}$. 
4. Integrate: $Ve^{-0.04t} = \frac{4000}{-0.02}e^{-0.02t} + C = -200,000e^{-0.02t} + C$. 
5. Isolate $V$: $V(t) = -200,000e^{0.02t} + Ce^{0.04t}$. 
6. IVP $V(0) = 10000 \implies 10000 = -200,000 + C \implies C = 210,000$. \\
\textbf{Final Answer:} $V(t) = 210,000e^{0.04t} - 200,000e^{0.02t}$

\textbf{Problem 28:} Mixture Model \\
\textbf{Solution:}
