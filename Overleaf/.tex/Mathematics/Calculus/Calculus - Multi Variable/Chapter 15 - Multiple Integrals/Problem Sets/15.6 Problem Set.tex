\textbf{Diagnostic Testing: "Find the Flaw"} \\
Identify the mathematical error in the flawed solution, explain it in one sentence, and provide the correct evaluation.
\begin{enumerate}
\setcounter{enumi}{27}
    \item \textbf{Evaluate:} $\displaystyle \int_0^1 \int_0^x \int_0^{y^2} 10z \,dz\,dy\,dx$ \\
    \textit{Flawed Solution:} Swapping $dx$ and $dy$ makes it easier. $\int_0^1 \int_0^{y^2} \int_0^x 10z \,dz\,dx\,dy = \int_0^1 \int_0^{y^2} 5x^2 y^2 \,dx\,dy = \int_0^1 \frac{5}{3} y^8 dy = \frac{5}{27}$.
    \item \textbf{Evaluate:} $\displaystyle \iiint_E x^2 \,dV$, over the cylinder $x^2+y^2 \le 4$, $0 \le z \le 3$. \\
    \textit{Flawed Solution:} In cylindrical coords, $\int_0^{2\pi} \int_0^2 \int_0^3 (r^2 \cos^2\theta) \,dz\,dr\,d\theta$. Inner is $3r^2\cos^2\theta$. Middle is $[r^3\cos^2\theta]_0^2 = 8\cos^2\theta$. Outer is $8(\pi) = 8\pi$.
    \item \textbf{Evaluate:} $\displaystyle \iiint_E z^3 \,dV$ over the box $E = [-1,1] \times [-1,1] \times [0,1]$. \\
    \textit{Flawed Solution:} The function $z^3$ is an odd function because $f(-z) = -f(z)$. Because we are integrating over a perfectly square box, the integral inherently cancels out to zero.
\end{enumerate}

\section*{Part 2: Detailed Solutions}

\textbf{1.} Separable constant bounds: $\int_0^2 x \,dx \cdot \int_0^1 y \,dy \cdot \int_0^3 4z^2 \,dz = [\frac{x^2}{2}]_0^2 \cdot [\frac{y^2}{2}]_0^1 \cdot [\frac{4z^3}{3}]_0^3 = (2) \cdot (\frac{1}{2}) \cdot (36) = 36$.

\textbf{2.} Split over the box: $\iiint 6x^2 = [\frac{6x^3}{3}]_{-1}^1 \times 2 \times 1 = (2 - (-2))(2) = 8$. $\iiint 4y = 2 \times [2y^2]_0^2 \times 1 = 2(8) = 16$. $\iiint z = 2 \times 2 \times [\frac{z^2}{2}]_0^1 = 4(1/2) = 2$. Total $= 8 + 16 + 2 = 26$.

\textbf{3.} Separable trick: $\int_0^{\pi/2} \cos(x) dx \cdot \int_0^\pi \sin(y) dy \cdot \int_0^1 z^2 dz = [\sin x]_0^{\pi/2} \cdot [-\cos y]_0^\pi \cdot [\frac{z^3}{3}]_0^1 = (1) \cdot (2) \cdot (\frac{1}{3}) = \frac{2}{3}$.

\textbf{4.} Inner ($dz$): $[120xz]_0^y = 120xy$. Middle ($dy$): $\int_0^x 120xy dy = [60xy^2]_0^x = 60x^3$. Outer ($dx$): $\int_0^1 60x^3 dx = [15x^4]_0^1 = 15$.

\textbf{5.} Inner ($dx$): $[3x^2]_0^z = 3z^2$. Middle ($dz$): $\int_0^y 3z^2 dz = [z^3]_0^y = y^3$. Outer ($dy$): $\int_0^2 y^3 dy = [\frac{y^4}{4}]_0^2 = 4$.

\textbf{6.} Inner ($dz$): $[6z]_0^{x-y} = 6(x-y)$. Middle ($dx$): $\int_y^1 6(x-y) dx = [3(x-y)^2]_y^1 = 3(1-y)^2$. Outer ($dy$): $\int_0^1 3(1-y)^2 dy$. U-sub $u=1-y, du=-dy \implies -\int_1^0 3u^2 du = [u^3]_0^1 = 1$.

\textbf{7.} Plane intercepts: $x=2, y=3, z=4$. Equation for $z$: $z = 4(1 - x/2 - y/3)$. $V = \int_0^2 \int_0^{3(1-x/2)} \int_0^{4(1-x/2-y/3)} 1 \,dz\,dy\,dx$.
Inner ($dz$): $4(1-x/2-y/3)$.
Middle ($dy$): $[4(1-x/2)y - \frac{2}{3}y^2]_0^{3(1-x/2)} = 12(1-x/2)^2 - 6(1-x/2)^2 = 6(1-x/2)^2$.
Outer ($dx$): $\int_0^2 6(1-x/2)^2 dx = 6[-\frac{2}{3}(1-x/2)^3]_0^2 = -4(0 - 1) = 4$.

\textbf{8.} Bounds: $z$ from $0$ to $4-2x-2y$. $y$ from $0$ to $2-x$. $x$ from $0$ to $2$.
Inner ($dz$): $\int_0^{4-2x-2y} 24z \,dz = [12z^2] = 12(4-2x-2y)^2 = 48(2-x-y)^2$.
Middle ($dy$): $\int_0^{2-x} 48(2-x-y)^2 dy$. Let $u=2-x-y$, $dy=-du$. Evaluate to $16(2-x)^3$.
Outer ($dx$): $\int_0^2 16(2-x)^3 dx = [-4(2-x)^4]_0^2 = 64$.

\textbf{9.} Bounds: $\int_{-1}^1 \int_0^2 \int_0^{1-x^2} (x+y) \,dz\,dy\,dx$.
Inner ($dz$): $(x+y)(1-x^2) = x-x^3+y-yx^2$.
Middle ($dy$): $[xy - x^3y + y^2/2 - x^2y^2/2]_0^2 = 2x - 2x^3 + 2 - 2x^2$.
Outer ($dx$): $\int_{-1}^1 (-2x^3 - 2x^2 + 2x + 2) dx$. Odd powers cancel due to symmetry. Even powers: $2 \int_0^1 (-2x^2 + 2) dx = 2[-\frac{2}{3}x^3 + 2x]_0^1 = 2(4/3) = \frac{8}{3}$.

\textbf{10.} Bounds: $\int_{-2}^2 \int_{x^2}^4 \int_0^3 1 \,dz\,dy\,dx$.
Inner ($dz$): $3$. Middle ($dy$): $\int_{x^2}^4 3 \,dy = 3(4-x^2) = 12-3x^2$.
Outer ($dx$): $\int_{-2}^2 (12-3x^2) dx = [12x - x^3]_{-2}^2 = (24-8) - (-24+8) = 32$.

\textbf{11.} \textbf{Answer: 0}. The domain $[-3,3]$ is symmetric across the $yz$-plane. The function $f(x,y,z) = x^3 \cos(z)$ is an odd function with respect to $x$ ($f(-x) = -f(x)$).

\textbf{12.} \textbf{Answer: 0}. The solid sphere is symmetric with respect to the $xz$-plane (bounds for $y$ are $-\sqrt{16-x^2-z^2}$ to $\sqrt{...}$). The integrand contains $y^5$, which is an odd function.

\textbf{13.} \textbf{Answer: 0}. The term $xy^2$ is odd wrt $x$, evaluated over $[-1,1]$. The term $x^2\sin(z)$ is odd wrt $z$, evaluated over $[-\sqrt{4-y^2}, \sqrt{4-y^2}]$. Both split integrals individually evaluate to zero.

\textbf{14.} Inner ($dz$): $[4yz\cos x]_0^x = 4xy \cos x$. Middle ($dy$): $\int_0^1 4xy \cos x \,dy = [2xy^2\cos x]_0^1 = 2x\cos x$.
Outer ($dx$): $\int_0^{\pi/2} 2x \cos x \,dx$. Use integration by parts: $u=2x, dv=\cos x dx$. $[2x\sin x + 2\cos x]_0^{\pi/2} = (\pi + 0) - (0 + 2) = \pi - 2$.

\textbf{15.} Inner ($dz$): $ye^y$. Middle ($dx$): $\int_0^y ye^y dx = [xye^y]_0^y = y^2e^y$.
Outer ($dy$): $\int_0^1 y^2e^y dy$. Parts twice ($u=y^2, dv=e^y dy$). $[y^2e^y - 2ye^y + 2e^y]_0^1 = (e-2e+2e) - (2) = e - 2$.

\textbf{16.} Inner ($dz$): $\int_0^{2-x-y} (2-x-y)^3 dz = [(2-x-y)^3 z]_0^{2-x-y} = (2-x-y)^4$.
Middle ($dy$): U-sub $u = 2-x-y \implies \int (2-x-y)^4 dy = [-\frac{1}{5}(2-x-y)^5]_0^{2-x} = \frac{1}{5}(2-x)^5$.
Outer ($dx$): U-sub $v = 2-x \implies \frac{1}{5} \int_0^2 (2-x)^5 dx = \frac{1}{5} [-\frac{1}{6}(2-x)^6]_0^2 = \frac{1}{30}(64) = \frac{32}{15}$.

\textbf{17.} Integrand is $r$. Volume element $dV = r\,dz\,dr\,d\theta$. Total integrand: $r^2$. Bounds: $z$ from $r^2$ to $4$, $r$ from $0$ to $2$.
Inner ($dz$): $\int_{r^2}^4 r^2 dz = r^2(4-r^2) = 4r^2 - r^4$.
Middle ($dr$): $\int_0^2 (4r^2 - r^4) dr = [\frac{4}{3}r^3 - \frac{1}{5}r^5]_0^2 = \frac{32}{3} - \frac{32}{5} = \frac{64}{15}$.
Outer ($d\theta$): $2\pi(\frac{64}{15}) = \frac{128\pi}{15}$.

\textbf{18.} Integrand is $1$. $dV = r\,dz\,dr\,d\theta$. Bounds: $z \in [r, 2], r \in [0, 2]$.
Inner: $r(2-r) = 2r - r^2$. Middle: $\int_0^2 (2r-r^2) dr = [r^2 - r^3/3]_0^2 = 4 - 8/3 = 4/3$. Outer: $2\pi(4/3) = \frac{8\pi}{3}$.

\textbf{19.} Orientation along $y$-axis: $x = r\cos\theta, z = r\sin\theta$. $dV = r \,dy\,dr\,d\theta$. Bounds $y \in [r^2, 9], r \in [0,3]$.
Inner ($dy$): $\int_{r^2}^9 yr \,dy = r[\frac{1}{2}y^2]_{r^2}^9 = \frac{r}{2}(81 - r^4)$.
Middle ($dr$): $\int_0^3 \frac{1}{2}(81r - r^5) dr = \frac{1}{2}[\frac{81}{2}r^2 - \frac{1}{6}r^6]_0^3 = \frac{1}{2}(\frac{729}{2} - \frac{729}{6}) = \frac{243}{2}$. Outer: $243\pi$.

\textbf{20.} Limits map to the first quadrant in the $xy$-plane ($\theta \in [0, \pi/2]$). Integrand $x^2+y^2 = r^2$. $dV = r \,dz\,dr\,d\theta$.
Inner: $\int_0^{9-r^2} r^3 dz = r^3(9-r^2) = 9r^3 - r^5$.
Middle: $\int_0^3 (9r^3 - r^5) dr = [\frac{9}{4}r^4 - \frac{1}{6}r^6]_0^3 = \frac{729}{4} - \frac{729}{6} = \frac{243}{4}$.
Outer: $\int_0^{\pi/2} \frac{243}{4} d\theta = \frac{243\pi}{8}$.

\textbf{21.} Spherical bounds: $0 \le \rho \le a, 0 \le \phi \le \pi, 0 \le \theta \le 2\pi$. $dV = \rho^2\sin\phi \,d\rho\,d\phi\,d\theta$.
Separable: $\int_0^a \rho^2 d\rho \int_0^\pi \sin\phi d\phi \int_0^{2\pi} 1 d\theta = (\frac{a^3}{3})(2)(2\pi) = \frac{4}{3}\pi a^3$.

\textbf{22.} Integrand is $\rho^2$. $dV = \rho^2\sin\phi$. Bounds: $\rho \in [0,2], \phi \in [0,\pi/2], \theta \in [0,2\pi]$.
Separable: $\int_0^2 \rho^4 d\rho \int_0^{\pi/2} \sin\phi d\phi \int_0^{2\pi} 1 d\theta = (\frac{32}{5})(1)(2\pi) = \frac{64\pi}{5}$.

\textbf{23.} Volume integrand $1$. Bounds $\rho \in [0,R], \phi \in [0,\pi/3], \theta \in [0,2\pi]$.
Separable: $\int_0^R \rho^2 d\rho \int_0^{\pi/3} \sin\phi d\phi \int_0^{2\pi} 1 d\theta = (\frac{R^3}{3}) \cdot (-\cos(\pi/3) - (-\cos(0))) \cdot 2\pi = \frac{R^3}{3}(1/2)(2\pi) = \frac{\pi R^3}{3}$.

\textbf{24.} Integrand $z = \rho\cos\phi$. Bounds $\rho \in [1,2], \phi \in [0,\pi/2], \theta \in [0,\pi/2]$.
Separable: $\int_1^2 \rho^3 d\rho \int_0^{\pi/2} \cos\phi\sin\phi d\phi \int_0^{\pi/2} d\theta = [\frac{\rho^4}{4}]_1^2 [\frac{\sin^2\phi}{2}]_0^{\pi/2} [\theta]_0^{\pi/2} = (\frac{15}{4})(\frac{1}{2})(\frac{\pi}{2}) = \frac{15\pi}{16}$.

\textbf{25.} Mass $= \iiint \rho(x,y,z) dV = \int_0^2 \int_0^2 \int_0^2 3x \,dz\,dy\,dx$.
Separable: $\int_0^2 3x \,dx \int_0^2 1 \,dy \int_0^2 1 \,dz = [3x^2/2]_0^2 \cdot (2) \cdot (2) = (6)(4) = 24$.

\textbf{26.} Volume $= 1 \times 2 \times 3 = 6$. Integral $= \int_0^1 x \,dx \int_0^2 y \,dy \int_0^3 z \,dz = (1/2)(2)(9/2) = 9/2$. Average $= \frac{9/2}{6} = \frac{3}{4}$.

\textbf{27.} $P = \int_1^\infty \int_1^\infty \int_1^\infty 6e^{-2x}e^{-y}e^{-3z} \,dz\,dy\,dx$.
Separable: $\int_1^\infty 2e^{-2x} dx \int_1^\infty e^{-y} dy \int_1^\infty 3e^{-3z} dz = [ -e^{-2x} ]_1^\infty [ -e^{-y} ]_1^\infty [ -e^{-3z} ]_1^\infty = e^{-2} \cdot e^{-1} \cdot e^{-3} = e^{-6}$.

\textbf{28.} \textbf{Error:} The student swapped the differentials $dx$ and $dy$ but improperly placed a variable ($x$) in the bounds of the outer $dx$ integral, violating the rule that outer limits must be constants. \\
\textit{Correct Solution:} Keep order $dz\,dy\,dx$. Inner: $[5z^2]_0^{y^2} = 5y^4$. Middle: $\int_0^x 5y^4 dy = x^5$. Outer: $\int_0^1 x^5 dx = 1/6$.

\textbf{29.} \textbf{Error:} The student forgot to multiply the integrand by the essential cylindrical Jacobian determinant, $r$. \\
\textit{Correct Solution:} Integrand must be $r^2\cos^2\theta \cdot r = r^3 \cos^2\theta$. $\int_0^{2\pi}\cos^2\theta d\theta \int_0^2 r^3 dr \int_0^3 dz = (\pi)(4)(3) = 12\pi$.

\textbf{30.} \textbf{Error:} The integration bounds for $z$ are $[0,1]$, which is entirely positive and not symmetric about the origin, so the odd-function cancellation rule cannot be applied. \\
\textit{Correct Solution:} Evaluate directly: $\int_{-1}^1 dx \int_{-1}^1 dy \int_0^1 z^3 dz = (2)(2)[z^4/4]_0^1 = 1$.
