\section*{1. Summations \& Statistics}
Sample Mean: 
$$ \bar{y} = \frac{1}{n}\sum_{i=1}^n y_i $$

Sample Variance:
$$ s^2 = \frac{1}{n-1}\sum_{i=1}^n (y_i - \bar{y})^2 $$
$$ s^2 = \frac{1}{n-1}\left[ \sum_{i=1}^n y_i^2 - \frac{1}{n}\left(\sum_{i=1}^n y_i\right)^2 \right] $$

Sample Standard Deviation:
$$ s = \sqrt{s^2} $$

Properties of Summations:
$$ \sum_{i=1}^n c = nc $$
$$ \sum_{i=1}^n c y_i = c \sum_{i=1}^n y_i $$
$$ \sum_{i=1}^n (x_i \pm y_i) = \sum_{i=1}^n x_i \pm \sum_{i=1}^n y_i $$
$$ \sum_{i=1}^n (y_i - \bar{y}) = 0 $$

Empirical Rule (Normal Distribution):
$$ \text{Interval } \mu \pm \sigma \approx 68\% $$
$$ \text{Interval } \mu \pm 2\sigma \approx 95\% $$
$$ \text{Interval } \mu \pm 3\sigma \approx 99.7\% $$

Z-Score (Standardization):
$$ Z = \frac{y - \bar{y}}{s} \quad \text{or} \quad Z = \frac{y - \mu}{\sigma} $$

\section*{2. Set Theory}
Commutative: 
$$A \cup B = B \cup A, \quad A \cap B = B \cap A$$

Associative: 
$$A \cup (B \cup C) = (A \cup B) \cup C$$
$$A \cap (B \cap C) = (A \cap B) \cap C$$

Distributive Laws:
$$ A \cap (B \cup C) = (A \cap B) \cup (A \cap C) $$
$$ A \cup (B \cap C) = (A \cup B) \cap (A \cup C) $$

De Morgan's Laws:
$$ \overline{A \cup B} = \bar{A} \cap \bar{B} $$
$$ \overline{A \cap B} = \bar{A} \cup \bar{B} $$

Partitioning a Set:
$$ A = (A \cap B) \cup (A \cap \bar{B}) $$

Mutually Exclusive (Disjoint):
$$ A \cap B = \emptyset $$

Absorption Law:
$$ B \cup (B \cap C) = B $$
$$ B \cap (B \cup C) = B $$

\section*{3. Probability Axioms}
Kolmogorov's Axioms:
$$ P(A) \ge 0 $$
$$ P(\Omega) = 1, \quad P(\emptyset) = 0 $$
If $A_i \cap A_j = \emptyset$ for $i \neq j$:
$$ P\left(\bigcup_{i=1}^\infty A_i\right) = \sum_{i=1}^\infty P(A_i) $$

Complement Rule:
$$ P(A) = 1 - P(\bar{A}) \iff P(A) + P(\bar{A}) = 1 $$

Monotonicity:
$$ A \subseteq B \implies P(A) \le P(B) $$

Discrete Uniform Measure:
$$ P(A) = \frac{|A|}{|\Omega|} $$

\section*{4. Combinatorics \& Counting}
$mn$ Rule (Fundamental Counting):
$$ |S| = n_1 \times n_2 \times \dots \times n_k $$

Permutations (Order matters):
$$ P_r^n = P(n, r) = \frac{n!}{(n-r)!} $$

Combinations (Order does not matter):
$$ C_r^n = \binom{n}{r} = \frac{n!}{r!(n-r)!} $$

Symmetry of Combinations:
$$ \binom{n}{r} = \binom{n}{n-r} $$

Pascal's Identity:
$$ \binom{n}{k} = \binom{n-1}{k} + \binom{n-1}{k-1} $$

Multinomial Partitions (Distinct items into groups):
$$ \binom{n}{n_1, n_2, \dots, n_k} = \frac{n!}{n_1! n_2! \dots n_k!} $$
$$\text{where } \sum_{i=1}^k n_i = n$$

Stars and Bars (Identical items into bins):
$$ \binom{n + k - 1}{k - 1} $$
$n$ = indistinguishable items, $k$ = distinguishable bins

Binomial Theorem:
$$ (x+y)^n = \sum_{k=0}^n \binom{n}{k} x^k y^{n-k} $$
$$ \sum_{k=0}^n \binom{n}{k} = 2^n $$

\section*{5. Addition \& Multiplication Rules}
Addition Rule (2 Events):
$$ P(A \cup B) = P(A) + P(B) - P(A \cap B) $$

If Mutually Exclusive ($A \cap B = \emptyset$):
$$ P(A \cup B) = P(A) + P(B) $$

Addition Rule (3 Events):
$$ P(A \cup B \cup C) = P(A) + P(B) + P(C) $$
$$ - P(A \cap B) - P(A \cap C) - P(B \cap C) $$
$$ + P(A \cap B \cap C) $$

Multiplicative Rule:
$$ P(A \cap B) = P(A|B)P(B) = P(B|A)P(A) $$

Multiplicative Rule (3 Events):
$$ P(A \cap B \cap C) = P(A) P(B|A) P(C|A \cap B) $$

\section*{6. Conditional Probability}
Definition:
$$ P(A|B) = \frac{P(A \cap B)}{P(B)}, \quad P(B) > 0 $$

Conditional Complement:
$$ P(\bar{A}|B) = 1 - P(A|B) $$

Conditional Addition Rule:
$$ P(A \cup B | C) = P(A|C) + P(B|C) - P(A \cap B | C) $$

Chain Rule for Complements:
$$ P(\bar{A} \cap \bar{B}) = P(\overline{A \cup B}) = 1 - P(A \cup B) $$

\section*{7. Independence}
Definition of Independent Events:
$$ P(A \cap B) = P(A)P(B) $$
$$ P(A|B) = P(A) $$
$$ P(B|A) = P(B) $$

If $A$ and $B$ are independent, so are:
$$ (A, \bar{B}), \quad (\bar{A}, B), \quad (\bar{A}, \bar{B}) $$

Mutual Independence (3 Events):
$$ P(A \cap B) = P(A)P(B) $$
$$ P(A \cap C) = P(A)P(C) $$
$$ P(B \cap C) = P(B)P(C) $$
$$ P(A \cap B \cap C) = P(A)P(B)P(C) $$

System Reliability (Independent Components):
$$ P(\text{Series}) = P(A_1 \cap A_2) = P(A_1)P(A_2) $$
$$ P(\text{Parallel}) = 1 - P(\bar{A_1} \cap \bar{A_2}) = 1 - P(\bar{A_1})P(\bar{A_2}) $$

\section*{8. Total Probability \& Bayes' Rule}
Law of Total Probability (LTP):
Given a partition $B_1, B_2, \dots, B_k$ of $\Omega$:
$$ P(A) = \sum_{i=1}^k P(A \cap B_i) = \sum_{i=1}^k P(A|B_i)P(B_i) $$

LTP (2 Events $B$ and $\bar{B}$):
$$ P(A) = P(A|B)P(B) + P(A|\bar{B})P(\bar{B}) $$

Bayes' Theorem:
$$ P(B_j|A) = \frac{P(A|B_j)P(B_j)}{\sum_{i=1}^k P(A|B_i)P(B_i)} $$

Bayes' Theorem (2 Events):
$$ P(B|A) = \frac{P(A|B)P(B)}{P(A|B)P(B) + P(A|\bar{B})P(\bar{B})} $$

Diagnostic Testing (Base Rates):
$D$ = Disease, $T$ = Positive Test
$$ \text{Sensitivity} = P(T|D) $$
$$ \text{Specificity} = P(\bar{T}|\bar{D}) $$
$$ PPV = P(D|T) = \frac{P(T|D)P(D)}{P(T|D)P(D) + P(T|\bar{D})P(\bar{D})} $$

\section*{9. Specific Distributions \& Bounds}
Hypergeometric Probability (from $N$ items, $K$ successes, sample $n$):
$$ P(X = k) = \frac{\binom{K}{k}\binom{N-K}{n-k}}{\binom{N}{n}} $$

Geometric Series (Infinite State Spaces):
$$ \sum_{k=0}^\infty a r^k = \frac{a}{1-r}, \quad |r| < 1 $$
$$ P(\text{A before B}) = \frac{P(A)}{P(A) + P(B)} $$

Boole's Inequality (Union Bound):
$$ P\left(\bigcup_{i=1}^n A_i\right) \le \sum_{i=1}^n P(A_i) $$

Bonferroni Inequality:
$$ P(A \cap B) \ge P(A) + P(B) - 1 $$
$$ P(A \cap B \cap C) \ge P(A) + P(B) + P(C) - 2 $$

Odds Ratio:
$$ \text{Odds}(A) = \frac{P(A)}{P(\bar{A})} = \frac{P(A)}{1 - P(A)} $$
